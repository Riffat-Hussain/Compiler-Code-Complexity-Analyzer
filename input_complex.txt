int main()
{
    int total = 0;
    int i = 0;

    for(i = 0; i < 20; i++)
    {
        if(i < 5)
        {
            total = total + i;
        }
        else if(i < 10)
        {
            for(int j = 0; j < i; j++)
            {
                if(j == 3)
                {
                    total = total - 1;
                }
            }
        }
        else
        {
            while(total > 0)
            {
                if(total == 42)
                {
                    total = 0;
                }
                total = total - 1;
            }
        }
    }

    return total;
}
